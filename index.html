package Lab_program3;
import java.util.Scanner;
public class Stud_Grade {
public static void main(String[] args) {
Scanner scanner = new Scanner(System.in);
// Step 1: Input student name
System.out.print("Enter Student Name: ");
String studentName = scanner.nextLine();
// Step 2: Input student percentage
System.out.print("Enter Student Percentage: ");
double percentage = scanner.nextDouble();
// Step 3: Verify and assign grade based on percentage
char grade;
if (percentage >= 90 && percentage <= 100) {
grade = 'A';
} else if (percentage >= 80 && percentage < 90) {
grade = 'B';
} else if (percentage >= 70 && percentage < 80) {
grade = 'C';
} else if (percentage >= 60 && percentage < 70) {
grade = 'D';
} else if (percentage < 50 && percentage >= 0) {
grade = 'F';
} else {

System.out.println("Invalid percentage entered.");
return;
}
// Step 4: Output student name, percentage, and grade
System.out.println("Student Name: " + studentName);
System.out.println("Student Percentage: " + percentage + "%");
System.out.println("Student Grade: " + grade);
scanner.close();
}
}

package Mytest;
import java.util.InputMismatchException;
import java.util.Scanner;
public class percentageabove60 {
public static void main(String[] args) {
Scanner scanner = new Scanner(System.in);
try {
// TC01 - Verify number of students
System.out.print("Enter the number of students: ");
int studentCount = scanner.nextInt();
scanner.nextLine();
if (studentCount <= 0) {
System.out.println("\nTC01 : FAIL");
System.out.println("Number of students must be greater than zero.");
return;
}
String[] names = new String[studentCount];
int[] percentages = new int[studentCount];
// Input Details
for (int i = 0; i < studentCount; i++) {
System.out.println("\nStudent Detail " + (i + 1));
System.out.print("Enter Name: ");
names[i] = scanner.nextLine();
System.out.print("Enter Percentage: ");

percentages[i] = scanner.nextInt();
scanner.nextLine();
}
System.out.println("\n========== TEST CASE VERIFICATION ==========");
// TC01
System.out.println("TC01 - Verify Number of Students : PASS");
// TC02
boolean validName = true;
for (String name : names) {
if (name.trim().isEmpty()) {
validName = false;
break;
}
}
if (validName)
System.out.println("TC02 - Verify Student Name : PASS");
else
System.out.println("TC02 - Verify Student Name : FAIL");
// TC03
boolean validPercentage = true;
for (int per : percentages) {
if (per < 0 || per > 100) {
validPercentage = false;
break;
}
}
if (validPercentage)
System.out.println("TC03 - Verify Percentage Range (0-100) : PASS");
else {
System.out.println("TC03 - Verify Percentage Range (0-100) : FAIL");
return;
}

// TC04
System.out.println("\nStudents with Percentage Above 60");
boolean found = false;
for (int i = 0; i < studentCount; i++) {
if (percentages[i] > 60) {
System.out.println(names[i] + " - " + percentages[i] + "%");
found = true;
}
}
if (found)
System.out.println("TC04 - Verify Students Above 60 : PASS");
else
System.out.println("TC04 - Verify Students Above 60 : FAIL (No Student Above 60)");
// TC05
System.out.println("TC05 - Verify Program Execution : PASS");
} catch (InputMismatchException e) {
System.out.println("\nTC06 - Verify Invalid Input : PASS");
System.out.println("Invalid input! Percentage must be numeric.");
} finally {
scanner.close();
}
}
